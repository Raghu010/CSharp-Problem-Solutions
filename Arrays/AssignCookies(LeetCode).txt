public class Solution 
{
    public int FindContentChildren(int[] g, int[] s) 
    {
        int count = 0;
        int iterator = 0;
        Array.Sort(g);
        Array.Sort(s);
        for (int i = 0; i < s.Length; i++) 
        {
            for (int j = iterator; j < g.Length; j++)
            {
                if (s[i] >= g[j]) 
                {
                    count += 1;
                    iterator += 1;
                    break;
                }
            }
        }
        return count;
    }
}
