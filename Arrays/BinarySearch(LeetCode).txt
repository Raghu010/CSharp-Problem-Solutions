public class Solution 
{
    public int Search(int[] nums, int target) 
    {
       int low = 0;
        int high = nums.Length;
        int index = -1;
        while (low < high) 
        {
            int mid = (low + high) / 2;
            if (nums[mid] == target)
                index = mid;
            if(target > nums[mid])
                low = mid + 1;
            else
                high = mid;
        }
        return index; 
    }
}
