public class Solution 
{
    public int CountPairs(IList<int> nums, int target) 
    {
        int output = 0;
        for (int i = 0; i < nums.Count; i++) 
        {
            int first = nums[i];
            for (int j = i + 1; j < nums.Count; j++) 
            {
                if(first + nums[j] <  target)
                    output = output + 1;
            }
        }
        return output;
    }
}
