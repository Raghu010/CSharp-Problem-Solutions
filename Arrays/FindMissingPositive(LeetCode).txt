public class Solution 
{
    public int FirstMissingPositive(int[] nums) 
    {
        Array.Sort(nums);
        int minimum = nums.Min();
        int maximum = nums.Max();
        int output = 0;
        if (minimum != 1 && (!nums.Contains(1)))
            output = 1;
        else
        {
            for (int i = 0; i < nums.Length; i++)
            {
                if (nums[i] > 0 && !nums.Contains(nums[i] + 1) && nums[i] != 0)
                {
                    output = nums[i] + 1;
                    if(output != 0)
                        break;
                }
            }
        }
        return output;
    }
}
