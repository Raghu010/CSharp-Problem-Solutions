public class Solution 
{
    public int HeightChecker(int[] heights) 
    {
       int[] sorted = (int[])heights.Clone();
        Array.Sort(sorted);
        int output = 0;
        for (int i = 0; i < sorted.Length; i++) 
        {
            if (heights[i] != sorted[i])
                output = output + 1;
        }
        return output; 
    }
}
