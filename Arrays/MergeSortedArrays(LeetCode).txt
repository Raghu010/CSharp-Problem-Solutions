public void Merge(int[] nums1, int m, int[] nums2, int n) 
    {
        int index = 0;
        if(!(n == 0) && !(m == 0))
        {
            for (int i = (m+index); i < nums1.Length; i++)
            {
                nums1[i] = nums2[index];
                index = index + 1;
            }
            Array.Sort(nums1);
        }
        else
        {
            for (int i = 0; i < n; i++)
            {
                nums1[i] = nums2[i];
            }
            Array.Sort(nums1);
        }
    }
