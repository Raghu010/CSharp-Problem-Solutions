public class Solution 
{
    public int MissingNumber(int[] nums) 
    {
        Array.Sort(nums);
        int output = nums.Length;
        if(!(nums[0] == 0))
        {
            output = 0;
        }
        else
        {
            for (int i = 0; i < nums.Length; i++)
            {
                if (i != nums[i])
                {
                    output = i;
                    break;
                }
            }
        }
        return output;
    }
}
