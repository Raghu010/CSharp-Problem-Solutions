public class Solution 
{
    public int NumOfPairs(string[] nums, string target) 
    {
        int result = 0;

        for (int i = 0; i < nums.Length; i++) 
        {
            string initial = nums[i];
            for (int j = 0; j < nums.Length; j++) 
            {
                if((i != j) && (initial +  nums[j] == target))
                    result = result + 1;
            }
        }
        return result;
    }
}
