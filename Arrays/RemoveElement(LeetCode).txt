public class Solution 
{
    public int RemoveElement(int[] nums, int val) 
    {
        int index = 0;
        int result = 0;
        for (int i = 0; i < nums.Length; i++)
        {
            if (nums[i] != val)
            {
                nums[index] = nums[i];
                index += 1;
                result += 1;
            }
        }
        return result;
    }
}
