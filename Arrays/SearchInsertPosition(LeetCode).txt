public class Solution 
{
    public int SearchInsert(int[] nums, int target) 
    {
        int index = 0;
        if (nums.Contains(target))
        {
            index = Array.IndexOf(nums, target);
        }
        else
        {
            for (int i = 0; i < nums.Length; i++) 
            {
                if (target > nums[i])
                {
                    index = index + 1;
                }
            }
        }
        return index;
    }
}
