public class Solution 
{
    public long MinimumSteps(string s) 
    {
        int whitePosition = 0;
        long totalSwaps = 0;
        for (int i = 0; i < s.Length; i++) 
        {
            if (s[i] == '0')
            {
                totalSwaps = totalSwaps + i - whitePosition;
                whitePosition = whitePosition + 1;
            }
        }
        return totalSwaps;
    }
}
