public class Solution 
{
    public int[] SortByBits(int[] arr) 
    {
        List<int> count = new List<int>();
        for (int i = 0; i < arr.Length; i++) 
        {
            int onesCount = Convert.ToString(arr[i], 2).Count(c => c ==  '1');
            count.Add(onesCount);
        }
        int[] sorted = arr.Select((num, index) => new { Value = num, Ones = count[index] }).OrderBy(x => x.Ones).ThenBy(x => x.Value).Select(x => x.Value).ToArray();
        return sorted;
    }
}
