public int[] TwoSum(int[] nums, int target)
{
    List<int> output = new List<int>();
    for (int i = 0; i < nums.Length; i++)
    {
        for (int j = i + 1; j < nums.Count(); j++)
        {
            if (nums[i] + nums[j] == target)
            {
                output.Add(i);
                output.Add(j);
            }
        }
    }
    return output.ToArray();
}
