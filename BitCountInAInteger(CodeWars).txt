/*Problem statement: Count the number of 1's in a given integer, after converting it to a bits*/

using System;
using System.Text.RegularExpressions;
using System.Collections.Generic;
using System.Linq;
using System.Web;
using System.Configuration;
using System.Data.SqlClient;
using static System.Net.Mime.MediaTypeNames;

namespace Practice
{
    
    public class Kata
    {
        public static int CountBits(int n)
        {
            string bits = Convert.ToString(n, 2);
            int count = 0;
            foreach (var s in bits)
            {
                if(int.Parse(s.ToString()) == 1)
                {
                    count = count + 1;
                }
            }
            return count;
        }

        static void Main()
        {
            int input = int.Parse(Console.ReadLine());
            Console.WriteLine(Kata.CountBits(input));
            Console.ReadKey();
        }
    }
    
}
