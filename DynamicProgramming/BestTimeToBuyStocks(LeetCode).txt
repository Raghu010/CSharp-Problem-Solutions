public class Solution 
{
    public int MaxProfit(int[] prices) 
    {
        int minBuy = int.MaxValue;
        int profit = 0;

        foreach (int price in prices) 
        {
            if (minBuy > price) 
            {
                minBuy = price;
            }
            else
            {
                profit = Math.Max(profit, price - minBuy);
            }
        }
        return profit;
    }
}
