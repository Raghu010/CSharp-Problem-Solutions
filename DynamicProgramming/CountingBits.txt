public class Solution 
{
    public int[] CountBits(int n) 
    {
        List<int> ints = new List<int>();
        string bits = "";
        for (int i = 0; i <= n; i++) 
        {
            int count = 0;
            bits = Convert.ToString(i, 2);
            foreach (var bit in bits) 
            {
                if (bit == '1')
                {
                    count = count + 1;
                }
            }
            ints.Add(count);
        }
        return ints.ToArray();
    }
}
