class Result
{
    /*Problem statement: Given an integer input, convert that to a binary and flip the bits and output the result in integer or a long value.*/
    public static long flippingBits(long n)
    {
        char[] bitsArr = Convert.ToString(n, 2).PadLeft(32, '0').ToArray();
        string output = "";
        foreach (var item in bitsArr)
        {
            if (item.Equals('0'))
            {
                output = output + "1";
            }
            else if(item.Equals('1'))
            {
                output = output + "0";
            }
        }
        ulong result = Convert.ToUInt64(output, 2);
        return (long)result;
    }
}

class Solution
{
    public static void Main(string[] args)
    {

        int q = Convert.ToInt32(Console.ReadLine().Trim());

        for (int qItr = 0; qItr < q; qItr++)
        {
            long n = Convert.ToInt64(Console.ReadLine().Trim());

            long result = Result.flippingBits(n);
            Console.WriteLine(result);
            Console.ReadKey();
        }
    }
}
