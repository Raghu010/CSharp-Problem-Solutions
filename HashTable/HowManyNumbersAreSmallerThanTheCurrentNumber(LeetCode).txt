public class Solution 
{
    public int[] SmallerNumbersThanCurrent(int[] nums) 
    {
        List<int> sorted = nums.ToList();
        sorted.Sort();
        int[] output = new int[nums.Length];
        for (int i = 0; i < nums.Length; i++) 
        {
            output[i] = sorted.IndexOf(nums[i]);
        }
        return output;
    }
}
