public class Solution 
{
    public string ClearDigits(string s) 
    {
        if (string.IsNullOrEmpty(s))
            return "";
        StringBuilder sb = new StringBuilder();
        foreach (char c in s)
        {
            if(char.IsDigit(c) && sb.Length > 0)
                sb.Length--;
            else
                sb.Append(c);
        }
        return sb.ToString();
    }
}
