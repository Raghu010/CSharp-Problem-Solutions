public class Solution 
{
    public string FindCommonResponse(IList<IList<string>> responses) 
    {
        List<string> distinct = new List<string>();
        Dictionary<string, int> dict = new Dictionary<string, int>();
        for (int i = 0; i < responses.Count; i++) 
        {
            foreach(var item in responses[i].Distinct())
            {
                distinct.Add(item);
            }
        }

        for (int i = 0; i < distinct.Count; i++) 
        {
            if (dict.ContainsKey(distinct[i]))
                dict[distinct[i]]++;
            else
                dict[distinct[i]] = 1;
        }

        var maxVal = dict.OrderByDescending(kvp => kvp.Value).ThenBy(kvp => kvp.Key).FirstOrDefault();
        return maxVal.Key;
    }
}
