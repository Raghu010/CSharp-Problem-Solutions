public class Solution 
{
    public IList<string> FizzBuzz(int n) 
    {
        IList<string> strList = new List<string>();
        for (int i = 1; i <= n; i++)
        {
            if ((i % 3 == 0 && i%5 == 0))
            {
                strList.Add("FizzBuzz");
            }
            else if (i % 3 == 0)
            {
                strList.Add("Fizz");
            }
            else if (i % 5 == 0) 
            {
                strList.Add("Buzz");
            }
            else
            {
                strList.Add(i.ToString());
            }
        }
        return strList;
    }
}
