public class Solution 
{
    public int FindComplement(int num) 
    {
        string bin = Convert.ToString(num, 2);
        string complement = "";
        foreach (var c in bin)
        {
            if (c == '0')
                complement = complement + '1';
            else
                complement = complement + '0';
        }
        return Convert.ToInt32(complement, 2);
    }
}
