public class Solution 
{
    public int ReverseDegree(string s) 
    {
        int result = 0;
        for (int i = 0; i < s.Length; i++) 
        {
            int multiplicationInteger = 'z' - s[i] + 1;
            result = result + (multiplicationInteger * (i+1));
        }
        return result;
    }
}
