public class Solution 
{
    public bool JudgeCircle(string moves) 
    {
        bool output = false;
        int up = 0;
        int down = 0;
        int left = 0;
        int right = 0;

        for (int i = 0; i < moves.Length; i++) 
        {
            if (moves[i] == 'U')
                up = up + 1;
            else if(moves[i] == 'D')
                down = down + 1;
            else if (moves[i] == 'L')
                left = left + 1;
            else if(moves[i] == 'R')
                right = right + 1;
        }

        if (up != 0 && down != 0 && up != 0 && down != 0) 
        {
            if (up == down && right == left)
                output = true;
        }
        if(left == 0 &&  right == 0)
        {
            if(up == down)
                output = true;
        }
        if(up == 0 && down == 0)
        {
            if(right == left)
                output = true;
        }
        return output;
    }
}
